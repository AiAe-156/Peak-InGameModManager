AssemblyNameReference Ref(string name, int major = 1, byte[]? token = null) => new(name, new Version(major, 0, 0, 0)) { PublicKeyToken = token ?? Array.Empty<byte>() };
void Dll(string directory, string name, AssemblyNameReference[]? references = null, bool mce = false, int major = 1, byte[]? token = null)
{
    Directory.CreateDirectory(directory);
    using var assembly = AssemblyDefinition.CreateAssembly(new AssemblyNameDefinition(name, new Version(major, 0, 0, 0)) { PublicKey = token ?? Array.Empty<byte>() }, name, ModuleKind.Dll);
    if (mce)
    {
        var module = assembly.MainModule;
        var type = new TypeDefinition("Fixture", "Plugin", TypeAttributes.Public | TypeAttributes.Class);
        module.Types.Add(type);
        var ctor = new MethodReference(".ctor", module.TypeSystem.Void, new TypeReference("BepInEx", "BepInPlugin", module, module)) { HasThis = true };
        for (int i = 0; i < 3; i++) ctor.Parameters.Add(new ParameterDefinition(module.TypeSystem.String));
        var attr = new CustomAttribute(ctor);
        foreach (string value in new[] { ComponentDependencyGraph.MceGuid, "MCE", "2.0.0" }) attr.ConstructorArguments.Add(new CustomAttributeArgument(module.TypeSystem.String, value));
        type.CustomAttributes.Add(attr);
    }
    assembly.MainModule.AssemblyReferences.Clear();
    foreach (var reference in references ?? Array.Empty<AssemblyNameReference>()) assembly.MainModule.AssemblyReferences.Add(reference);
    assembly.Write(Path.Combine(directory, name + ".dll"));
}
